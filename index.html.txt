    <script>
        // Default Storage Initialization
        let defaultUsers = [
            { id: "BM-101", name: "করিম আহমেদ", role: "সেলস ম্যানিয়াগার", pass: "1234", type: "staff" },
            { id: "BM-102", name: "শফিক ইসলাম", role: "মেকানিক", pass: "1234", type: "staff" },
            { id: "admin", name: "শোরুম ম্যানেজার", role: "অ্যাডমিন", pass: "1234", type: "admin" },
            { id: "owner", name: "মালিক (বাচ্চু মটরস)", role: "সুপার অ্যাডমিন", pass: "1234", type: "superadmin" }
        ];

        if (!localStorage.getItem("bm_users")) {
            localStorage.setItem("bm_users", JSON.stringify(defaultUsers));
        }

        if (!localStorage.getItem("bm_attendance")) {
            localStorage.setItem("bm_attendance", JSON.stringify([]));
        }

        let currentUser = null;
        let capturedLocation = null;
        let capturedPhotoData = null;

        // Current Date Display
        document.getElementById("currentDateDisplay").innerText = new Date().toLocaleDateString('bn-BD', {
            year: 'numeric', month: 'long', day: 'numeric'
        });

        // Authentication Handler
        function handleLogin(e) {
            e.preventDefault();
            const uInput = document.getElementById("username").value.trim();
            const pInput = document.getElementById("password").value.trim();
            
            const users = JSON.parse(localStorage.getItem("bm_users"));
            const user = users.find(u => u.id.toLowerCase() === uInput.toLowerCase() && u.pass === pInput);

            if (user) {
                currentUser = user;
                showPanel(user);
            } else {
                alert("ইউজার আইডি বা পাসওয়ার্ড ভুল হয়েছে!");
            }
        }

        function showPanel(user) {
            document.getElementById("loginSection").classList.add("hidden");
            document.getElementById("userBadge").classList.remove("hidden");
            document.getElementById("roleTag").innerText = user.role;

            if (user.type === "staff") {
                document.getElementById("staffPanel").classList.remove("hidden");
                document.getElementById("staffNameDisplay").innerText = user.name;
                document.getElementById("staffIdDisplay").innerText = user.id;
                document.getElementById("staffRoleDisplay").innerText = user.role;
                loadStaffHistory();
            } else if (user.type === "admin") {
                document.getElementById("adminPanel").classList.remove("hidden");
                loadAdminView();
            } else if (user.type === "superadmin") {
                document.getElementById("superAdminPanel").classList.remove("hidden");
                loadSuperAdminView();
            }
        }

        function logout() {
            location.reload();
        }

        // Camera & GPS Functions
        function startCamera() {
            const video = document.getElementById("webcam");
            const placeholder = document.getElementById("cameraPlaceholder");
            
            navigator.mediaDevices.getUserMedia({ video: true })
                .then(stream => {
                    video.srcObject = stream;
                    video.classList.remove("hidden");
                    placeholder.classList.add("hidden");
                })
                .catch(err => {
                    alert("ক্যামেরা অন করা যাচ্ছে না। পারমিশন চেক করুন।");
                });
        }

        function capturePhoto() {
            const video = document.getElementById("webcam");
            const preview = document.getElementById("photoPreview");
            if (!video.srcObject) return alert("প্রথমে ক্যামেরা অন করুন!");

            const canvas = document.createElement("canvas");
            canvas.width = 300;
            canvas.height = 200;
            const ctx = canvas.getContext("2d");
            ctx.drawImage(video, 0, 0, 300, 200);

            capturedPhotoData = canvas.toDataURL("image/jpeg");
            preview.src = capturedPhotoData;
            preview.classList.remove("hidden");
            video.classList.add("hidden");
        }

        function getLocation() {
            const status = document.getElementById("locationStatus");
            if (navigator.geolocation) {
                status.innerText = "লোকেশন খোঁজা হচ্ছে...";
                navigator.geolocation.getCurrentPosition(
                    pos => {
                        capturedLocation = {
                            lat: pos.coords.latitude,
                            lng: pos.coords.longitude
                        };
                        status.innerText = `লোকেশন পাওয়া গেছে (Lat: ${pos.coords.latitude.toFixed(4)})`;
                        status.classList.add("text-emerald-600");
                    },
                    err => {
                        status.innerText = "লোকেশন পারমিশন পাওয়া যায়নি!";
                        status.classList.add("text-red-600");
                    }
                );
            }
        }

        // Submit Attendance
        function submitAttendance() {
            if (!capturedPhotoData) {
                alert("অনুগ্রহ করে ক্যামেরা দিয়ে ছবি তুলুন!");
                return;
            }

            const overtime = parseFloat(document.getElementById("overtimeHours").value) || 0;
            const records = JSON.parse(localStorage.getItem("bm_attendance"));

            const newRecord = {
                staffId: currentUser.id,
                staffName: currentUser.name,
                timestamp: new Date().toLocaleString("bn-BD"),
                dateStr: new Date().toLocaleDateString(),
                photo: capturedPhotoData,
                location: capturedLocation,
                overtime: overtime
            };

            records.push(newRecord);
            localStorage.setItem("bm_attendance", JSON.stringify(records));
            alert("আপনার হাজিরা সফলভাবে জমা হয়েছে!");
            loadStaffHistory();
        }

        // Data Population Functions
        function loadStaffHistory() {
            const records = JSON.parse(localStorage.getItem("bm_attendance"));
            const myRecords = records.filter(r => r.staffId === currentUser.id);
            const table = document.getElementById("staffHistoryTable");

            if (myRecords.length === 0) {
                table.innerHTML = `<tr><td colspan="4" class="p-4 text-center text-slate-400">কোনো হিস্ট্রি পাওয়া যায়নি</td></tr>`;
                return;
            }

            table.innerHTML = myRecords.map(r => `
                <tr class="border-b">
                    <td class="p-2">${r.dateStr}</td>
                    <td class="p-2">${r.timestamp.split(',')[1] || r.timestamp}</td>
                    <td class="p-2">${r.overtime} ঘণ্টা</td>
                    <td class="p-2">${r.location ? `<a href="https://maps.google.com/?q=${r.location.lat},${r.location.lng}" target="_blank" class="text-blue-600 underline text-xs">ম্যাপ দেখুন</a>` : 'N/A'}</td>
                </tr>
            `).join('');
        }

        function loadAdminView() {
            const users = JSON.parse(localStorage.getItem("bm_users"));
            const records = JSON.parse(localStorage.getItem("bm_attendance"));
            const todayStr = new Date().toLocaleDateString();

            const staffOnly = users.filter(u => u.type === 'staff');
            const todayRecords = records.filter(r => r.dateStr === todayStr);
            const totalOT = todayRecords.reduce((sum, r) => sum + r.overtime, 0);

            document.getElementById("adminTotalStaff").innerText = staffOnly.length;
            document.getElementById("adminTodayAttendance").innerText = todayRecords.length;
            document.getElementById("adminTodayOvertime").innerText = totalOT;

            const table = document.getElementById("adminAttendanceTable");
            if (records.length === 0) {
                table.innerHTML = `<tr><td colspan="5" class="p-4 text-center text-slate-400">কোনো হাজিরা ডাটা নেই</td></tr>`;
                return;
